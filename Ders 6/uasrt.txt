/* USER CODE BEGIN 2 */

// Timer 2 modülünü PWM modunda başlatır.
// Bu komut olmazsa sayaç çalışmaz, LED hep 0V kalır.
HAL_TIM_PWM_Start(&htim2, TIM_CHANNEL_1);

/* USER CODE END 2 */



/* USER CODE BEGIN 3 */

// --- 1. NEFES ALMA (Parlaklık Artıyor) ---
// i değişkeni bizim "Gaz Pedalımız"dır. 0'dan başlar, 1000'e kadar çıkar.
// Neden 1000? Çünkü .ioc ayarlarında 'Counter Period' değerini 999 yaptık.
for(int i = 0; i <= 1000; i += 10)
{
    // EN KRİTİK KOMUT: CCR (Capture Compare Register) değerini değiştirir.
    // Timer donanımına der ki: "Sayacın 0 ile 1000 arası sayarken,
    // i değerine kadar elektriği AÇIK tut, sonrasında KAPAT."
    // i=100 ise -> %10 Açık kalır.
    // i=900 ise -> %90 Açık kalır.
    __HAL_TIM_SET_COMPARE(&htim2, TIM_CHANNEL_1, i);

    // Göz Yanılması İçin Bekleme
    // Bu bekleme olmazsa 0'dan 1000'e mikrosaniyede çıkar, biz hep yanıyor sanırız.
    // 10ms bekleyerek yavaşça parladığını görüyoruz.
    HAL_Delay(10);
}

// Zirvede (Tam Parlak) biraz bekle
HAL_Delay(200);


// --- 2. NEFES VERME (Parlaklık Azalıyor) ---
// Gaz pedalını 1000'den (Tam gaz) geriye doğru 0'a çekiyoruz.
for(int i = 1000; i >= 0; i -= 10)
{
    // Yeni düşük değeri Timer'a yükle
    __HAL_TIM_SET_COMPARE(&htim2, TIM_CHANNEL_1, i);

    // Geçişi görmek için bekle
    HAL_Delay(10);
}

// Sönük halde biraz bekle
HAL_Delay(500);

/* USER CODE END 3 */

Clock Source: Internal Clock yap.

Channel 1: PWM Generation CH1 seç. (PA5 pini yeşil olacak).

Aşağıdaki Configuration (Parameter Settings) Kısmı (KRİTİK!):

İşlemcimiz 72 MHz (72.000.000 Hz) hızında koşuyor. Biz bunu yavaşlatıp 1 kHz (1000 Hz) PWM elde etmek istiyoruz.

Prescaler (PSC): 71 yap. (Formül: 72MHz / (71+1) = 1 MHz Sayaç Hızı).

Counter Period (ARR): 999 yap. (Formül: 1 MHz / (999+1) = 1000 Hz = 1 kHz).

Özetle: Bizim hız kadranımız 0 ile 1000 arasında olacak.
